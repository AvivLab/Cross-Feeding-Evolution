\pgfplotsset{
  mccp cond combined/.style={
    width=\mccpcondplotw,
    height=\mccpcondploth,
    xmin=-1, xmax=5,
    ymin=0, ymax=1.12,
    ytick={0,0.2,0.4,0.6,0.8,1},
    xlabel={$E_i$},
    xlabel style={font=\scriptsize, yshift=0pt},
    ylabel={Probability},
    ylabel style={font=\scriptsize},
    tick label style={font=\scriptsize},
    grid=major,
    grid style={gray!20},
    clip=false,
  },
  mccp death/.style={orange!85!black, thick, const plot, mark=none},
  mccp dup ref/.style={blue!55!black, thick, const plot, mark=none},
  mccp dup primary/.style={blue!70!black, thick, const plot, mark=none},
  mccp dup coupled/.style={blue!60!black, thick, const plot, mark=none},
  mccp outflow/.style={gray!55!black, thick, dashed, const plot, mark=none},
}

\def\mccpcondphi{0.2}
\def\mccpconddupflow{0.25} % phi/(1-phi) at phi=0.2
\newcommand{\mccpcondaddoutflow}{%
  \addplot[mccp outflow]
    coordinates {(-1,\mccpcondphi) (5,\mccpcondphi)};
}

% Per-panel local bbox gives each subplot's own left edge (incl.\ ylabel).
\newcommand{\mccpcondpaneldecor}[1]{%
  \coordinate (mccpplotleft) at (mccppanel.south west);
  \node[
    anchor=south west,
    inner sep=0pt,
    outer sep=0pt,
  ] (mccpcondheader) at ([yshift=\mccpcondheadsep]#1.north west) {%
    \parbox[t][\mccpcondheaderh][b]{\mccpcondheadtextw}{%
      {\footnotesize\bfseries \mccpcondtitle}\\[0.02em]
      {\scriptsize \mccpcondsubtitle}%
    }%
  };
  \node[
    anchor=south west,
    font=\footnotesize\bfseries,
    inner sep=0pt,
    outer sep=0pt,
  ] at ([yshift=\mccpcondlabelpad]mccpplotleft |- mccpcondheader.north) {(\mccpcondletter)};
}

\begin{scope}[shift={(\mccpcondxshift,0)}]

% --- (a) Neutral ---
\begin{scope}[local bounding box=mccppanel, shift={(0,\dimexpr\mccpcondbottombase+\mccpcondrowpitch\relax)}]
  \begin{axis}[
    mccp cond combined,
    name=axA,
    at={(0,\mccpcondxlabelhang)}, anchor=south west,
  ]
    \addplot[mccp death]
      coordinates {(-1,0) (5,0)};
    \addplot[mccp dup coupled]
      coordinates {(-1,\mccpconddupflow) (5,\mccpconddupflow)};
    \mccpcondaddoutflow
  \end{axis}
  \def\mccpcondletter{a}%
  \def\mccpcondtitle{Neutral}%
  \def\mccpcondsubtitle{No Death; Flow-Balanced Duplication}%
  \mccpcondpaneldecor{axA}
\end{scope}

% --- (b) Differential Death ---
\begin{scope}[local bounding box=mccppanel, shift={(\mccpcondcolB,\dimexpr\mccpcondbottombase+\mccpcondrowpitch\relax)}]
  \begin{axis}[
    mccp cond combined,
    name=axB,
    at={(0,\mccpcondxlabelhang)}, anchor=south west,
  ]
    \addplot[mccp death]
      coordinates {(-1,1) (0,1)};
    \addplot[orange!85!black, thick, domain=0:5, samples=120, forget plot]
      {exp(-2*x)};
    \addplot[mccp dup ref]
      coordinates {(-1,0.3) (5,0.3)};
    \mccpcondaddoutflow
  \end{axis}
  \def\mccpcondletter{b}%
  \def\mccpcondtitle{Differential Death}%
  \def\mccpcondsubtitle{eq.\ (4); Constant Duplication}%
  \mccpcondpaneldecor{axB}
\end{scope}

% --- (c) Differential Reproduction ---
\begin{scope}[local bounding box=mccppanel, shift={(0,\mccpcondbottombase)}]
  \begin{axis}[
    mccp cond combined,
    name=axC,
    at={(0,\mccpcondxlabelhang)}, anchor=south west,
  ]
    \addplot[mccp death]
      coordinates {(-1,0) (5,0)};
    \addplot[mccp dup primary]
      coordinates {(-1,0) (0,0) (0,{1/(1+exp(12.5))})};
    \addplot[blue!70!black, thick, domain=0:5, samples=120, forget plot]
      {1/(1+exp(-5*(x-2.5)))};
    \mccpcondaddoutflow
  \end{axis}
  \def\mccpcondletter{c}%
  \def\mccpcondtitle{Differential Reproduction}%
  \def\mccpcondsubtitle{No Death; eq.\ (5)}%
  \mccpcondpaneldecor{axC}
\end{scope}

% --- (d) Differential Death and Duplication ---
