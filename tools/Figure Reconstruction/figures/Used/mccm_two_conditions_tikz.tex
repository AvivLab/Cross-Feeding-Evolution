\def\mccmcondphi{0.2}
\def\mccmconddupflow{0.25}
\newcommand{\mccmcondaddoutflow}{%
  \addplot[mccm outflow]
    coordinates {(-1,\mccmcondphi) (5,\mccmcondphi)};
}

\newcommand{\mccmcondpaneldecor}[1]{%
  \coordinate (mccmplotleft) at (mccmpanel.south west);
  \node[
    anchor=south west,
    inner sep=0pt,
    outer sep=0pt,
  ] (mccmcondheader) at ([yshift=\mccmcondheadsep]#1.north west) {%
    \parbox[t][\mccmcondheaderh][b]{\mccmcondheadtextw}{%
      {\footnotesize\bfseries \mccmcondtitle}\\[0.02em]
      {\scriptsize \mccmcondsubtitle}%
    }%
  };
  \node[
    anchor=south west,
    font=\footnotesize\bfseries,
    inner sep=0pt,
    outer sep=0pt,
  ] at ([yshift=\mccmcondlabelpad]mccmplotleft |- mccmcondheader.north) {(\mccmcondletter)};
}

\begin{scope}[shift={(\mccmcondxshift,0)}]

% --- (a) Neutral ---
\begin{scope}[local bounding box=mccmpanel, shift={(0,\mccmcondbottombase)}]
  \begin{axis}[

  \def\mccmcondletter{b}%
  \def\mccmcondtitle{Selection Regime}%
  \def\mccmcondsubtitle{eq.\ (4), (5)}%
