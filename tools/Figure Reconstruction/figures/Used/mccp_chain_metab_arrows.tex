% Measure half-width (square) or radius (circle) from the rendered node shape.
\def\mccpchainbound@measure#1#2{%
  \pgfpointanchor{#1}{east}\pgf@xa=\pgf@x
  \pgfpointanchor{#1}{center}\pgf@xc=\pgf@x
  \pgfmathsetmacro{#2}{(\pgf@xa-\pgf@xc)/1cm}%
}
% L-infinity boundary on an axis-aligned square: #2=(...) target coordinate expression.
\def\mccpchainbound@square@i#1#2#3{%
  \coordinate (mccp@bnd@vec) at ($ (#2) - (#1.center) $);%
  \pgf@process{\pgfpointanchor{mccp@bnd@vec}{center}}%
  \pgfmathsetmacro{\mccp@dx}{\pgf@x/1cm}%
  \pgfmathsetmacro{\mccp@dy}{\pgf@y/1cm}%
  \pgfmathparse{abs(\mccp@dx)>=abs(\mccp@dy) ? (\mccp@dx>=0 ? 1 : -1) : (\mccp@dy>=0 ? 2 : -2)}%
  \pgfmathtruncatemacro{\mccp@anchnum}{\pgfmathresult}%
  \ifnum\mccp@anchnum=1 \def\mccp@anch{east}\else
    \ifnum\mccp@anchnum=-1 \def\mccp@anch{west}\else
      \ifnum\mccp@anchnum=2 \def\mccp@anch{north}\else
        \def\mccp@anch{south}\fi\fi\fi
  \coordinate (#3) at (#1.\mccp@anch);%
}
\def\mccpchainbound@square#1#2#3#4{%
  \mccpchainbound@square@i{#1}{#2.center}{#4}%
}
\def\mccpchainbound@squaretoc#1#2#3#4{%
  \mccpchainbound@square@i{#1}{#2}{#4}%
}
\def\mccpchainbound@circle#1#2#3#4{%
  \mccpchainbound@measure{#1}{\mccp@r}%
  \coordinate (#4) at ($(#1.center)!{\mccp@r cm}!(#2.center)$);%
}
\def\mccpchainbound@circletoc#1#2#3#4{%
  \mccpchainbound@measure{#1}{\mccp@r}%
  \coordinate (#4) at ($(#1.center)!{\mccp@r cm}!(#2)$);%
}
\def\mccpchainboundpoint#1#2#3#4#5{%
  \def\mccpchainbound@shape{#4}%
  \ifx\mccpchainbound@shape circle
    \mccpchainbound@circle{#1}{#2}{#3}{#5}%
  \else
    \mccpchainbound@square{#1}{#2}{#3}{#5}%
  \fi
}
\def\mccpchainboundpointtoc#1#2#3#4#5{%
  \def\mccpchainbound@shape{#4}%
  \ifx\mccpchainbound@shape circle
    \mccpchainbound@circletoc{#1}{#2}{#3}{#5}%
  \else
    \mccpchainbound@squaretoc{#1}{#2}{#3}{#5}%
  \fi
}

% #1=segment index, #2=1 on final chunk (body stops at bodystop)
\def\mccpchainmetab@segloc#1#2#3{%
  \ifnum#2=1
    \pgfmathsetmacro{\mccpchainlocA}{max(0, (#1-\mccpchainsegoverlap)*\mccpchainarrowcolorstop/\mccpchaingradsteps)}%
    \pgfmathsetmacro{\mccpchainlocB}{min(\mccpchainarrowcolorstop, (#1+1)*\mccpchainarrowcolorstop/\mccpchaingradsteps)}%
  \else
    \pgfmathsetmacro{\mccpchainlocA}{max(0, (#1-\mccpchainsegoverlap)/\mccpchaingradsteps)}%
    \pgfmathsetmacro{\mccpchainlocB}{min(1, (#1+1)/\mccpchaingradsteps)}%
  \fi
}
% Draw one gradient chunk; #1--#2 endpoints, #3=path offset in [0,1), #4=1 on final chunk
\def\mccpchaindrawmetab@chunk#1#2#3#4{%
  \ifnum#4=1
    \foreach \j in {0,1,...,\mccpchaingradlast} {%
      \mccpchainmetab@segloc{\j}{1}{}%
      \pgfmathsetmacro{\mccpchainuA}{#3+\mccpchainlocA/4}%
      \pgfmathsetmacro{\mccpchainuB}{#3+\mccpchainlocB/4}%
      \coordinate (mbMetPtA) at ($(#1)!{\mccpchainlocA}!(#2)$);%
      \coordinate (mbMetPtB) at ($(#1)!{\mccpchainlocB}!(#2)$);%
      \pgfmathsetmacro{\mccpchainmixval}{int(round(100*(1-0.5*(\mccpchainuA+\mccpchainuB))))}%
      \edef\mccpchainmixexp{\mccpchainmixval}%
      \ifnum\j=\mccpchaingradlast
        \def\mccpchaincapline{mccpchainmetabline@butt}%
      \else
        \def\mccpchaincapline{mccpchainmetabline}%
      \fi
      \draw[\mccpchaincapline,
        draw=mccpchaincA!\mccpchainmixexp!mccpchaincB] (mbMetPtA) -- (mbMetPtB);%
    }%
    \coordinate (mbMetShaftEnd) at ($(#1)!\mccpchainarrowshaftstop!(#2)$);%
    \mccpchaindrawmetab@shaftbridge{#1}{#2}{mccpchaincB}%
    \mccpchaindrawmetab@splithead{#1}{mbMetShaftEnd}{#2}{mccpchaincB};%
  \else
    \foreach \j in {0,1,...,\mccpchaingradlast} {%
      \mccpchainmetab@segloc{\j}{0}{}%
      \pgfmathsetmacro{\mccpchainuA}{#3+\mccpchainlocA/4}%
      \pgfmathsetmacro{\mccpchainuB}{#3+\mccpchainlocB/4}%
      \coordinate (mbMetPtA) at ($(#1)!{\mccpchainlocA}!(#2)$);%
      \coordinate (mbMetPtB) at ($(#1)!{\mccpchainlocB}!(#2)$);%
      \pgfmathsetmacro{\mccpchainmixval}{int(round(100*(1-0.5*(\mccpchainuA+\mccpchainuB))))}%
      \edef\mccpchainmixexp{\mccpchainmixval}%
      \draw[mccpchainmetabline,
        draw=mccpchaincA!\mccpchainmixexp!mccpchaincB] (mbMetPtA) -- (mbMetPtB);%
    }%
  \fi
}

% On-circle lane anchors: #1=org node, #2/#3=in/out coord names, #4=radius (cm), #5=height fraction.
\def\mccpchainorglanecoords#1#2#3#4#5{%
  \pgfmathsetmacro{\mccp@lanef}{#5}%
  \pgfmathsetmacro{\mccp@laney}{\mccp@lanef* (#4)}%
  \pgfmathsetmacro{\mccp@lanex}{sqrt(1-\mccp@lanef*\mccp@lanef)*(#4)}%
  \coordinate (#2) at ($(#1.center)+({-\mccp@lanex cm},{\mccp@laney cm})$);%
  \coordinate (#3) at ($(#1.center)+({\mccp@lanex cm},{\mccp@laney cm})$);%
}
% Dual-task lane: route through org above/below center (closer to midline than metabolite rows).
\def\mccpchainorglanefrac{0.48}  % |lane height| as fraction of org radius
\def\mccpchaindrawmetabarrowLane#1#2#3#4#5#6#7{%
  \colorlet{mccpchaincA}{#5}%
  \colorlet{mccpchaincB}{#6}%
  \makeatletter
  \pgfpointanchor{#3}{center}\pgf@ya=\pgf@y
  \pgfpointanchor{#2}{center}\pgf@yb=\pgf@y
  \pgfmathsetmacro{\mccp@dy}{(\pgf@yb-\pgf@ya)/1cm}%
  \makeatother
  \pgfmathsetmacro{\mccp@lanef}{\mccpchainorglanefrac}%
  \ifdim\mccp@dy pt<0pt
    \pgfmathsetmacro{\mccp@lanef}{-\mccp@lanef}%
  \fi
  \mccpchainbound@measure{#3}{\mccp@orgr}%
  \pgfmathsetmacro{\mccp@lanex}{sqrt(1-\mccp@lanef*\mccp@lanef)*\mccp@orgr}%
  \pgfmathsetmacro{\mccp@laney}{\mccp@lanef*\mccp@orgr}%
  \coordinate (#1@In) at ($(#3.center)+({-\mccp@lanex cm},{\mccp@laney cm})$);%
  \coordinate (#1@Out) at ($(#3.center)+({\mccp@lanex cm},{\mccp@laney cm})$);%
  \coordinate (#1@Mid) at ($(#3.center)+(0 cm,{\mccp@laney cm})$);%
  \mccpchainboundpointtoc{#2}{#1@In}{0}{square}{#1@Src}%
  \mccpchainboundpoint{#4}{#3}{0}{square}{#1@Snk}%
  \draw[mccpchainmetabline, shorten <=0.6pt, draw=mccpchaincA]
    (#1@Src) -- (#1@In);%
  \mccpchaindrawmetab@chunk{#1@In}{#1@Mid}{0.25}{0}%
  \mccpchaindrawmetab@chunk{#1@Mid}{#1@Out}{0.5}{0}%
  \mccpchaindrawmetab@chunk{#1@Out}{#1@Snk}{0.75}{1}%
  \mccpchainmetab@joint{#1@In}{mccpchaincA!88!mccpchaincB}%
  \mccpchainmetab@joint{#1@Mid}{mccpchaincA!50!mccpchaincB}%
  \mccpchainmetab@joint{#1@Out}{mccpchaincA!12!mccpchaincB}%
}
